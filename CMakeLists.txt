cmake_minimum_required(VERSION 3.16)
project(LanDesk VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets Network)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets Network)
set(LANDESK_HOST_MOC "" CACHE FILEPATH "Host Qt moc for cross compilation (must match target Qt major/minor)")
if(CMAKE_CROSSCOMPILING AND LANDESK_HOST_MOC)
  set_property(TARGET Qt${QT_VERSION_MAJOR}::moc PROPERTY IMPORTED_LOCATION "${LANDESK_HOST_MOC}")
endif()
find_package(OpenSSL REQUIRED)
add_library(landesk_core STATIC src/protocol.cpp src/protocol.h src/native_input.cpp src/native_input.h src/app.cpp src/app.h src/screen_capture.cpp src/screen_capture.h src/video_codec.cpp src/video_codec.h src/video_pipeline.cpp src/video_pipeline.h)
target_include_directories(landesk_core PUBLIC src)
target_sources(landesk_core PRIVATE src/desktop_lock.cpp src/desktop_lock.h)
target_link_libraries(landesk_core PUBLIC Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Network OpenSSL::Crypto)
find_package(PkgConfig QUIET)
option(LANDESK_ENABLE_H264 "Enable FFmpeg/libx264 video when dependencies are available" ON)
if(LANDESK_ENABLE_H264 AND PkgConfig_FOUND)
  pkg_check_modules(FFMPEG QUIET IMPORTED_TARGET libavcodec libavutil libswscale)
  if(FFMPEG_FOUND)
    target_compile_definitions(landesk_core PUBLIC LANDESK_HAVE_FFMPEG)
    target_link_libraries(landesk_core PUBLIC PkgConfig::FFMPEG)
    message(STATUS "H.264 enabled; runtime requires libx264 encoder")
  endif()
endif()
add_executable(landesk src/main.cpp)
target_link_libraries(landesk PRIVATE landesk_core)
if(WIN32)
  set_target_properties(landesk PROPERTIES WIN32_EXECUTABLE TRUE)
  if(TARGET Qt${QT_VERSION_MAJOR}::WinMain)
    target_link_libraries(landesk PRIVATE Qt${QT_VERSION_MAJOR}::WinMain)
  endif()
  target_link_libraries(landesk_core PUBLIC user32 gdi32)
elseif(UNIX AND NOT APPLE)
  find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS DBus)
  target_link_libraries(landesk_core PUBLIC Qt${QT_VERSION_MAJOR}::DBus)
  find_package(X11 REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CAPTURE REQUIRED IMPORTED_TARGET xext xrandr xrender xcb xcb-shm)
  target_include_directories(landesk_core PRIVATE ${X11_INCLUDE_DIR})
  target_link_libraries(landesk_core PUBLIC ${X11_LIBRARIES} PkgConfig::CAPTURE)
else()
  message(FATAL_ERROR "LanDesk currently supports Windows and Linux/X11")
endif()
if(MSVC)
  target_compile_options(landesk_core PUBLIC /W4 /utf-8)
else()
  target_compile_options(landesk_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
option(LANDESK_BUILD_TESTS "Build interaction and invitation tests" OFF)
if(LANDESK_BUILD_TESTS)
  find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
  enable_testing()
  add_executable(landesk_tests tests/viewer_test.cpp)
  target_link_libraries(landesk_tests PRIVATE landesk_core Qt${QT_VERSION_MAJOR}::Test)
  add_test(NAME viewer COMMAND landesk_tests)
  if(UNIX AND NOT APPLE)
    add_executable(landesk_desktop_lock_tests tests/desktop_lock_test.cpp src/desktop_lock.cpp)
    target_include_directories(landesk_desktop_lock_tests PRIVATE src)
    target_link_libraries(landesk_desktop_lock_tests PRIVATE Qt${QT_VERSION_MAJOR}::DBus Qt${QT_VERSION_MAJOR}::Test)
    add_test(NAME desktop_lock COMMAND landesk_desktop_lock_tests)
  endif()
  if(LANDESK_ENABLE_H264 AND FFMPEG_FOUND)
    add_executable(landesk_codec_tests tests/video_codec_test.cpp)
    target_link_libraries(landesk_codec_tests PRIVATE landesk_core)
    add_test(NAME codec COMMAND landesk_codec_tests)
  endif()
endif()
install(TARGETS landesk RUNTIME DESTINATION bin)
