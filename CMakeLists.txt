cmake_minimum_required(VERSION 3.16)
project(bananaDesk VERSION 0.7.6 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets Network)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets Network)
set(LANDESK_HOST_MOC "" CACHE FILEPATH "Host Qt moc for cross compilation (must match target Qt major/minor)")
set(LANDESK_HOST_RCC "" CACHE FILEPATH "Host Qt rcc for cross compilation")
if(CMAKE_CROSSCOMPILING AND LANDESK_HOST_MOC)
  set_property(TARGET Qt${QT_VERSION_MAJOR}::moc PROPERTY IMPORTED_LOCATION "${LANDESK_HOST_MOC}")
endif()
if(CMAKE_CROSSCOMPILING AND LANDESK_HOST_RCC)
  set_property(TARGET Qt${QT_VERSION_MAJOR}::rcc PROPERTY IMPORTED_LOCATION "${LANDESK_HOST_RCC}")
endif()
find_package(OpenSSL REQUIRED)
add_library(landesk_core STATIC src/protocol.cpp src/protocol.h src/native_input.cpp src/native_input.h src/app.cpp src/app.h src/screen_capture.cpp src/screen_capture.h src/video_codec.cpp src/video_codec.h src/video_pipeline.cpp src/video_pipeline.h)
target_include_directories(landesk_core PUBLIC src)
target_sources(landesk_core PRIVATE src/desktop_lock.cpp src/desktop_lock.h)
target_sources(landesk_core PRIVATE src/desktop_wake.cpp src/desktop_wake.h)
target_sources(landesk_core PRIVATE src/file_transfer.cpp src/file_transfer.h src/file_transfer_dialog.cpp src/file_transfer_dialog.h)
target_sources(landesk_core PRIVATE src/clipboard_sync.cpp src/clipboard_sync.h)
target_sources(landesk_core PRIVATE src/identity_store.cpp src/identity_store.h)
target_sources(landesk_core PRIVATE src/device_store.cpp src/device_store.h)
target_link_libraries(landesk_core PUBLIC Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Network OpenSSL::Crypto)
find_package(PkgConfig QUIET)
option(LANDESK_ENABLE_H264 "Enable FFmpeg/libx264 video when dependencies are available" ON)
if(LANDESK_ENABLE_H264 AND PkgConfig_FOUND)
  pkg_check_modules(FFMPEG QUIET IMPORTED_TARGET libavcodec libavutil libswscale)
  if(FFMPEG_FOUND)
    target_compile_definitions(landesk_core PUBLIC LANDESK_HAVE_FFMPEG)
    target_link_libraries(landesk_core PUBLIC PkgConfig::FFMPEG)
    message(STATUS "H.264 enabled; runtime requires libx264 encoder")
  endif()
endif()
add_executable(landesk src/main.cpp assets/branding.qrc)
set_target_properties(landesk PROPERTIES OUTPUT_NAME bananaDesk AUTORCC ON)
target_link_libraries(landesk PRIVATE landesk_core)
if(WIN32)
  enable_language(RC)
  configure_file(windows/bananaDesk.rc.in "${CMAKE_CURRENT_BINARY_DIR}/bananaDesk.rc" @ONLY)
  set_property(SOURCE "${CMAKE_CURRENT_BINARY_DIR}/bananaDesk.rc" PROPERTY OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/bananaDesk.ico")
  target_sources(landesk PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/bananaDesk.rc")
  set_target_properties(landesk PROPERTIES WIN32_EXECUTABLE TRUE)
  if(TARGET Qt${QT_VERSION_MAJOR}::WinMain)
    target_link_libraries(landesk PRIVATE Qt${QT_VERSION_MAJOR}::WinMain)
  endif()
  target_link_libraries(landesk_core PUBLIC user32 gdi32)
  target_sources(landesk_core PRIVATE src/windows_bridge.cpp src/windows_bridge.h)
  target_link_libraries(landesk_core PUBLIC advapi32 shell32 crypt32)
  add_executable(landesk-helper windows/helper_main.cpp)
  set_target_properties(landesk-helper PROPERTIES AUTOMOC OFF)
  target_compile_definitions(landesk-helper PRIVATE UNICODE _UNICODE)
  target_link_libraries(landesk-helper PRIVATE user32 gdi32 advapi32 shell32 wtsapi32 bcrypt)
  if(MINGW)
    target_link_options(landesk-helper PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
  install(TARGETS landesk-helper RUNTIME DESTINATION bin)
elseif(UNIX AND NOT APPLE)
  find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS DBus)
  target_link_libraries(landesk_core PUBLIC Qt${QT_VERSION_MAJOR}::DBus)
  find_package(X11 REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CAPTURE REQUIRED IMPORTED_TARGET xext xrandr xrender xcb xcb-shm)
  target_include_directories(landesk_core PRIVATE ${X11_INCLUDE_DIR})
  target_link_libraries(landesk_core PUBLIC ${X11_LIBRARIES} PkgConfig::CAPTURE)
else()
  message(FATAL_ERROR "bananaDesk currently supports Windows and Linux/X11")
endif()
if(MSVC)
  target_compile_options(landesk_core PUBLIC /W4 /utf-8)
else()
  target_compile_options(landesk_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
option(LANDESK_BUILD_TESTS "Build interaction and invitation tests" OFF)
if(LANDESK_BUILD_TESTS)
  find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
  enable_testing()
  add_executable(landesk_identity_tests tests/identity_store_test.cpp)
  target_link_libraries(landesk_identity_tests PRIVATE landesk_core Qt${QT_VERSION_MAJOR}::Test)
  add_test(NAME identity_store COMMAND landesk_identity_tests)
  add_executable(landesk_persistent_sharing_tests tests/persistent_sharing_test.cpp)
  target_link_libraries(landesk_persistent_sharing_tests PRIVATE landesk_core Qt${QT_VERSION_MAJOR}::Test)
  add_test(NAME persistent_sharing COMMAND landesk_persistent_sharing_tests)
  add_executable(landesk_proxy_tests tests/proxy_connection_test.cpp)
  target_link_libraries(landesk_proxy_tests PRIVATE landesk_core Qt${QT_VERSION_MAJOR}::Test)
  add_test(NAME proxy_connection COMMAND landesk_proxy_tests)
  add_executable(landesk_clipboard_tests tests/clipboard_sync_test.cpp src/clipboard_sync.cpp)
  target_include_directories(landesk_clipboard_tests PRIVATE src)
  target_link_libraries(landesk_clipboard_tests PRIVATE Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Test)
  add_test(NAME clipboard COMMAND landesk_clipboard_tests)
  add_executable(landesk_tests tests/viewer_test.cpp)
  target_link_libraries(landesk_tests PRIVATE landesk_core Qt${QT_VERSION_MAJOR}::Test)
  add_test(NAME viewer COMMAND landesk_tests)
  add_executable(landesk_file_client_tests tests/file_transfer_client_test.cpp)
  target_link_libraries(landesk_file_client_tests PRIVATE landesk_core Qt${QT_VERSION_MAJOR}::Test)
  add_test(NAME file_client COMMAND landesk_file_client_tests)
  add_executable(landesk_file_transfer_dialog_tests tests/file_transfer_dialog_test.cpp)
  target_link_libraries(landesk_file_transfer_dialog_tests PRIVATE landesk_core Qt${QT_VERSION_MAJOR}::Test)
  add_test(NAME file_dialog COMMAND landesk_file_transfer_dialog_tests)
  add_executable(landesk_file_offer_ui_tests tests/file_offer_ui_test.cpp)
  target_link_libraries(landesk_file_offer_ui_tests PRIVATE landesk_core Qt${QT_VERSION_MAJOR}::Test)
  add_test(NAME file_offer_ui COMMAND landesk_file_offer_ui_tests)
  add_executable(landesk_file_transfer_tests tests/file_transfer_test.cpp)
  target_link_libraries(landesk_file_transfer_tests PRIVATE landesk_core Qt${QT_VERSION_MAJOR}::Test)
  add_test(NAME file_transfer COMMAND landesk_file_transfer_tests)
  if(UNIX AND NOT APPLE)
    add_executable(landesk_clipboard_client_tests tests/clipboard_client_test.cpp)
    target_link_libraries(landesk_clipboard_client_tests PRIVATE landesk_core Qt${QT_VERSION_MAJOR}::Test)
    add_test(NAME clipboard_client COMMAND landesk_clipboard_client_tests)
    add_executable(landesk_desktop_lock_tests tests/desktop_lock_test.cpp src/desktop_lock.cpp)
    target_include_directories(landesk_desktop_lock_tests PRIVATE src)
    target_link_libraries(landesk_desktop_lock_tests PRIVATE Qt${QT_VERSION_MAJOR}::DBus Qt${QT_VERSION_MAJOR}::Test)
    add_test(NAME desktop_lock COMMAND landesk_desktop_lock_tests)
    add_executable(landesk_desktop_wake_tests tests/desktop_wake_test.cpp src/desktop_wake.cpp)
    target_include_directories(landesk_desktop_wake_tests PRIVATE src ${X11_INCLUDE_DIR})
    target_link_libraries(landesk_desktop_wake_tests PRIVATE Qt${QT_VERSION_MAJOR}::Test ${X11_LIBRARIES} PkgConfig::CAPTURE ${X11_Xss_LIB})
    add_test(NAME desktop_wake COMMAND landesk_desktop_wake_tests)
    add_executable(landesk_unlock_protocol_tests tests/unlock_protocol_test.cpp)
    target_link_libraries(landesk_unlock_protocol_tests PRIVATE landesk_core Qt${QT_VERSION_MAJOR}::Test)
    add_test(NAME unlock_protocol COMMAND landesk_unlock_protocol_tests)
  endif()
  if(LANDESK_ENABLE_H264 AND FFMPEG_FOUND)
    add_executable(landesk_codec_tests tests/video_codec_test.cpp)
    target_link_libraries(landesk_codec_tests PRIVATE landesk_core)
    add_test(NAME codec COMMAND landesk_codec_tests)
  endif()
endif()
install(TARGETS landesk RUNTIME DESTINATION bin)
if(UNIX AND NOT APPLE)
  install(FILES assets/bananaDesk.png DESTINATION share/icons/hicolor/512x512/apps)
  install(FILES linux/bananaDesk.desktop DESTINATION share/applications)
  install(PROGRAMS
    linux/install-prelogin.sh
    linux/uninstall-prelogin.sh
    linux/bananaDesk-prelogin-xauth
    linux/bananaDesk-prelogin-launch
    DESTINATION share/bananaDesk/linux)
endif()
